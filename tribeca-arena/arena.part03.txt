  function frame(ts){
    const dt=Math.min(.05,(ts-state.lastFrame)/1000||0);state.lastFrame=ts;
    if(state.started){
      if(state.isHost){simulate(dt);if(ts-state.lastSnapshot>SNAPSHOT_MS){state.lastSnapshot=ts;send('state',buildSnapshot());updateHud();updateScoreboard();}}
      else if(ts-state.lastInputSend>INPUT_MS){state.lastInputSend=ts;send('input',{id:state.id,input:{...state.localInput}});updateHud();}
      render();
    } else renderAttract(ts);
    requestAnimationFrame(frame);
  }
